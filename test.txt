public class TransactionAnalyzerTests
{
    private const string Ok = "Транзакция допустима.";
    private const string MsgAmount = "Ошибка: сумма должна быть положительной.";
    private const string MsgUnknown = "Ошибка: неизвестный тип транзакции.";
    private const string MsgCount = "Ограничение: превышено количество операций в день.";
    private const string MsgTotal = "Ограничение: превышена дневная сумма транзакций.";
    private const string MsgAtm = "Ограничение: банкоматы не обрабатывают крупные суммы.";
    private const string MsgExternal = "Отклонено: крупные внешние переводы недопустимы.";
    private const string MsgCommission = "Комиссия: 1% за перевод между типами счетов.";
    private const string MsgWeekend = "Ошибка: в выходные нельзя переводить между разными типами счетов через онлайн-каналы.";
    private const string MsgWeb = "Отклонено: снятие через веб-банкинг запрещено.";
    private const string MsgWithdrawal = "Ограничение: крупное снятие только для VIP-клиентов.";

    private const string Wed = "2026-10-07";
    private const string Sat = "2026-10-10";
    private const string Sun = "2026-10-11";

    private static string MakeTransaction(
        int amount = 1000, TransactionKind kind = TransactionKind.Deposit, bool vip = false,
        string channel = "Office", bool isInternal = true, string from = "A", string to = "A",
        int count = 0, int total = 0, string day = Wed)
        => TransactionAnalyzer.AnalyzeTransaction(new Transaction
        {
            Amount = amount,
            Kind = kind,
            IsVipClient = vip,
            Channel = channel,
            IsInternal = isInternal,
            FromAccountType = from,
            ToAccountType = to,
            DailyTransactionCount = count,
            DailyTransactionTotal = total,
            Timestamp = DateTime.Parse(day, CultureInfo.InvariantCulture)
        });

    [Theory]
    [InlineData(0, TransactionKind.Deposit, MsgAmount)]
    [InlineData(100, TransactionKind.Unknown, MsgUnknown)]
    public void AmountOrKindInvalid_ReturnsError(int amount, TransactionKind kind, string expected) =>
        Assert.Equal(expected, MakeTransaction(amount: amount, kind: kind));

    [Theory]
    [InlineData(10, false, Ok)]
    [InlineData(11, false, MsgCount)]
    [InlineData(11, true, Ok)]
    public void DailyCount_RespectsLimitAndVip(int count, bool vip, string expected) =>
        Assert.Equal(expected, MakeTransaction(count: count, vip: vip));

    [Theory]
    [InlineData(500_000, 1_000_000, false, Ok)]
    [InlineData(500_001, 1_000_000, false, MsgTotal)]
    [InlineData(500_000, 2_000_000, true, Ok)]
    public void DailyTotal_RespectsLimitAndVip(int amount, int total, bool vip, string expected) =>
        Assert.Equal(expected, MakeTransaction(amount: amount, total: total, vip: vip));

    [Theory]
    [InlineData(100_000, Ok)]
    [InlineData(100_001, MsgAtm)]
    public void AtmChannel_RespectsAmountLimit(int amount, string expected) =>
        Assert.Equal(expected, MakeTransaction(amount: amount, channel: "ATM"));

    [Theory]
    [InlineData(1_000_000, false, false, "A", "A", Ok)]
    [InlineData(1_000_001, false, false, "A", "A", MsgExternal)]
    [InlineData(1_400_000, true, false, "A", "A", Ok)]
    [InlineData(1_000_001, false, true, "A", "A", Ok)]
    [InlineData(1_000, false, false, "A", "B", MsgCommission)]
    [InlineData(1_000, false, true, "A", "B", Ok)]
    [InlineData(1_000, true, false, "A", "B", Ok)]
    public void Transfer_RespectsLimitsCommissionAndVip(int amount, bool isInternal, bool vip, string from, string to, string expected) =>
        Assert.Equal(expected, MakeTransaction(amount: amount, kind: TransactionKind.Transfer,
            isInternal: isInternal, vip: vip, from: from, to: to));

    [Theory]
    [InlineData(Sat, "Web", "A", "B", MsgWeekend)]
    [InlineData(Sun, "Mobile", "A", "B", MsgWeekend)]
    [InlineData(Sat, "Office", "A", "B", Ok)]
    [InlineData(Sat, "Web", "A", "A", Ok)]
    [InlineData(Wed, "Web", "A", "B", Ok)]
    public void WeekendTransfer_RejectsOnlineBetweenDifferentTypes(string day, string channel, string from, string to, string expected) =>
        Assert.Equal(expected, MakeTransaction(kind: TransactionKind.Transfer, day: day, channel: channel, from: from, to: to));

    [Theory]
    [InlineData(1_000, true, false, "Web", MsgWeb)]
    [InlineData(200_000, false, false, "Office", Ok)]
    [InlineData(200_001, false, false, "Office", MsgWithdrawal)]
    [InlineData(300_000, true, false, "Office", Ok)]
    [InlineData(300_000, false, true, "Office", Ok)]
    public void Withdrawal_RespectsWebBanAndLimit(int amount, bool isInternal, bool vip, string channel, string expected) =>
        Assert.Equal(expected, MakeTransaction(amount: amount, kind: TransactionKind.Withdrawal,
            isInternal: isInternal, vip: vip, channel: channel));

    [Fact]
    public void Deposit_IgnoresTransferAndWithdrawalRules() =>
        Assert.Equal(Ok, MakeTransaction(amount: 1_200_000, isInternal: false, channel: "Web",
            from: "A", to: "B", day: Sat));

}
